import React, { useCallback, useEffect, useState } from 'react';
import { useNavigate, useParams } from 'react-router-dom';
import Button from '../../Components/Button';
import Card from '../../Components/Card';
import DashboardShell from '../../Components/dashboard/DashboardShell';
import StatusBadge from '../../Components/dashboard/StatusBadge';
import api from '../../api/axios';
import { CheckCircle, MapPin, Star, User } from '../../icons';
import { cn } from '../../lib/cn';

export default function ClientRequestDetails() {
  const { id } = useParams();
  const navigate = useNavigate();
  const [request, setRequest] = useState(null);
  const [loading, setLoading] = useState(true);
  const [rating, setRating] = useState(5);
  const [reviewComment, setReviewComment] = useState('');
  const [submittingReview, setSubmittingReview] = useState(false);

  const fetchRequest = useCallback(async () => {
    setLoading(true);
    try {
      const response = await api.get(`/requests/${id}`);
      setRequest(response.data.data ?? response.data);
    } finally {
      setLoading(false);
    }
  }, [id]);

  useEffect(() => {
    fetchRequest();
  }, [fetchRequest]);

  const handleSubmitReview = async (event) => {
    event.preventDefault();
    setSubmittingReview(true);
    try {
      await api.post(`/requests/${id}/reviews`, {
        rating,
        comment: reviewComment,
      });
      await fetchRequest();
    } finally {
      setSubmittingReview(false);
    }
  };

  const acceptedOffer = request?.offers?.find((offer) => offer.status === 'accepted');
  const selectedProvider = request?.selected_provider;
  const existingReview = request?.review;
  const providerName = selectedProvider
    ? `${selectedProvider.first_name ?? ''} ${selectedProvider.last_name ?? ''}`.trim()
    : acceptedOffer
    ? `${acceptedOffer.provider?.first_name ?? ''} ${acceptedOffer.provider?.last_name ?? ''}`.trim()
    : '';

  return (
    <DashboardShell role="client" title="Détail de la demande" description="Suivez l'avancement de votre mission.">
      {loading ? (
        <div className="rounded-lg border border-border bg-white p-8 text-center text-text-muted">Chargement...</div>
      ) : request ? (
        <div className="grid gap-8 lg:grid-cols-[2fr_1fr]">
          <div className="flex flex-col gap-8">
            <Card className="text-start">
              <div className="mb-4 flex flex-col gap-3 sm:flex-row sm:items-start sm:justify-between">
                <div>
                  <h2 className="text-2xl font-bold text-text-dark">{request.category?.name}</h2>
                  <p className="mt-2 flex items-center gap-1.5 text-sm text-text-muted"><MapPin size={15} /> {request.city}</p>
                </div>
                <StatusBadge status={request.status} />
              </div>
              <p className="whitespace-pre-wrap text-pretty leading-7 text-text-main">{request.description}</p>
              <div className="mt-6 border-t border-border pt-6">
                <p className="text-sm text-text-muted">Budget proposé</p>
                <p className="text-2xl font-bold text-primary tabular-nums">{request.proposed_price} DH</p>
              </div>
            </Card>

            {/* Mission accepted / in-progress banner */}
            {(request.status === 'in_progress' || acceptedOffer) && request.status !== 'completed' ? (
              <Card className="border-primary/20 bg-primary-light/30 text-start">
                <div className="flex items-center gap-2 font-bold text-primary">
                  <CheckCircle size={20} />
                  Mission acceptée par {providerName}
                </div>
                {request.status === 'in_progress' ? (
                  <p className="mt-2 text-sm text-text-main">Un talent travaille sur votre mission.</p>
                ) : null}
              </Card>
            ) : null}

            {/* Completed banner */}
            {request.status === 'completed' ? (
              <Card className="border-success/20 bg-success-bg text-start">
                <div className="flex items-center gap-2 font-bold text-success">
                  <CheckCircle size={20} />
                  Mission terminée
                </div>
              </Card>
            ) : null}

            {/* Review section: show form when completed and no review exists; show read-only review if one exists */}
            {request.status === 'completed' ? (
              existingReview ? (
                <Card className="text-start">
                  <h2 className="mb-4 text-xl font-bold text-text-dark">Votre avis</h2>
                  <div className="mb-3 flex items-center gap-1">
                    {[1, 2, 3, 4, 5].map((star) => (
                      <Star
                        key={star}
                        size={24}
                        fill={star <= existingReview.rating ? 'currentColor' : 'none'}
                        className={star <= existingReview.rating ? 'text-warning' : 'text-gray-300'}
                      />
                    ))}
                    <span className="ml-2 text-sm text-text-muted">({existingReview.rating}/5)</span>
                  </div>
                  {existingReview.comment ? (
                    <p className="whitespace-pre-wrap text-pretty leading-7 text-text-main">{existingReview.comment}</p>
                  ) : (
                    <p className="text-sm text-text-muted">Aucun commentaire.</p>
                  )}
                </Card>
              ) : (
                <Card className="border-primary/20 bg-primary-light/30 text-start">
                  <h2 className="mb-4 text-xl font-bold text-text-dark">Évaluer la prestation</h2>
                  <form onSubmit={handleSubmitReview} className="flex flex-col gap-4">
                    <div>
                      <label className="mb-2 block text-sm font-semibold text-text-dark">Note</label>
                      <div className="flex gap-2">
                        {[1, 2, 3, 4, 5].map((star) => (
                          <button
                            key={star}
                            type="button"
                            onClick={() => setRating(star)}
                            className={cn('transition-colors duration-200', rating >= star ? 'text-warning' : 'text-gray-300')}
                          >
                            <Star size={32} fill="currentColor" />
                          </button>
                        ))}
                      </div>
                    </div>
                    <div>
                      <label className="mb-2 block text-sm font-semibold text-text-dark">Commentaire</label>
                      <textarea
                        value={reviewComment}
                        onChange={(event) => setReviewComment(event.target.value)}
                        className="w-full rounded-sm border border-border px-4 py-3 outline-none focus:border-primary focus:ring-4 focus:ring-accent/20"
                        rows="3"
                      />
                    </div>
                    <Button type="submit" disabled={submittingReview} className="self-start">
                      {submittingReview ? 'Envoi...' : "Soumettre l'avis"}
                    </Button>
                  </form>
                </Card>
              )
            ) : null}

            {/* Offers list (read-only) */}
            <div className="text-start">
              <h2 className="mb-6 text-xl font-bold text-text-dark">Offres reçues ({request.offers?.length || 0})</h2>
              <div className="flex flex-col gap-4">
                {request.offers?.map((offer) => {
                  const offerProvider = selectedProvider && offer.status === 'accepted'
                    ? selectedProvider
                    : offer.provider;
                  const offerProviderName = `${offerProvider?.first_name ?? ''} ${offerProvider?.last_name ?? ''}`.trim();
                  const isAccepted = offer.status === 'accepted';

                  return (
                    <Card key={offer.id} className={cn('text-start', isAccepted && 'border-success ring-1 ring-success')}>
                      <div className="flex flex-col gap-4 sm:flex-row sm:items-center sm:justify-between">
                        <div className="flex items-center gap-4">
                          <div className="flex size-12 items-center justify-center rounded-sm bg-primary/10 font-bold text-primary">
                            {offerProvider?.first_name?.[0]}{offerProvider?.last_name?.[0]}
                          </div>
                          <div>
                            <p className="font-bold text-text-dark">{offerProviderName}</p>
                            <p className="text-xs text-text-muted">{offerProvider?.completed_jobs || 0} missions</p>
                          </div>
                        </div>
                        <div className="text-left sm:text-right">
                          {offer.offered_price ? (
                            <p className="text-xl font-bold text-primary tabular-nums">{offer.offered_price} DH</p>
                          ) : null}
                          <StatusBadge status={offer.status} />
                        </div>
                      </div>
                      {offer.message ? <p className="mt-4 text-sm text-text-muted">{offer.message}</p> : null}
                      <div className="mt-4 flex flex-wrap gap-3">
                        <Button variant="outline" className="px-4 py-2 text-xs" onClick={() => navigate(`/talents/${offer.provider_id}`)}>
                          <User size={14} /> Voir profil
                        </Button>
                      </div>
                    </Card>
                  );
                })}
                {request.offers?.length === 0 ? (
                  <div className="rounded-lg border border-dashed border-border bg-white p-8 text-center text-text-muted">
                    Aucune offre pour le moment.
                  </div>
                ) : null}
              </div>
            </div>
          </div>

          <Card className="h-fit text-start">
            <h3 className="mb-3 font-bold text-text-dark">Conseil de sécurité</h3>
            <p className="text-sm leading-6 text-text-muted">
              Le talent accepté vous contactera via WhatsApp. Gardez les accords de prix et de livraison clairs.
            </p>
          </Card>
        </div>
      ) : (
        <div className="rounded-lg border border-border bg-white p-8 text-center text-text-muted">Demande introuvable.</div>
      )}
    </DashboardShell>
  );
}
